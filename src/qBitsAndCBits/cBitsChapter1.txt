# Define basic single-bit matrices
n <- matrix(c(0,0,0,1), nrow=2, byrow=TRUE)
ntilde <- matrix(c(1,0,0,0), nrow=2, byrow=TRUE)
x <- matrix(c(0,1,1,0), nrow=2, byrow=TRUE)

# Tensor (Kronecker) products
nA_nB <- kronecker(n, n)
ntildeA_ntildeB <- kronecker(ntilde, ntilde)
xA_xB <- kronecker(x, x)

# Verify SWAP operator
swap <- ntildeA_ntildeB + nA_nB +
        (kronecker(ntilde, n) + kronecker(n, ntilde)) %*% xA_xB
