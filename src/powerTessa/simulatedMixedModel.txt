
rm(list=ls())

library("nlme")


rSquareds <- vector()
pdf("c:\\temp\\sims.pdf")

par(mfrow=c(3,2))

allX <- vector()
allY <- vector()
allPatients <- vector()

# 4 drug classes evenly distributed
# TODO: A more accurate power simulation would allow for different sized classes

allDrug <- vector()
index <- 1
drugIndex <- 1

for( i in 1:150) 
{
	x <- c(-7,0,7,14,21,28,35,42)

	# choose slope so that true R^2 ≈ 0.25
	slope <- 1
	sigma <- sqrt((slope^2 * var(x)) * (1 - 0.5) / 0.5)
	
	y <- slope * x + rnorm(length(x), sd = sigma)
	
	if( drugIndex == 1 ) 
		y = y + y * .25
	
	rSquareds[i] =  summary(lm(y ~ x))$r.squared

	plot(x,y, main = rSquareds[i] )
	
	for( j in 1:length(x))
	{
		allX[index] <- x[j]
		allY[index] <- y[j]
		allPatients[index] <- i
		allDrug[index] <- drugIndex		
		index <- index + 1
	}
	
	drugIndex <- drugIndex + 1
		
	if(drugIndex == 5) 
		drugIndex =1 ;

}

#should be around .25
mean(rSquareds)
dev.off()

#x is time; y is response (tolerance)
myFrame <- data.frame( allX, allY, allDrug, allPatients)

M.mixed <- lme( allY ~ allX + allDrug, method= "REML", random = ~1 | allPatients, data = myFrame)

anova(M.mixed)
