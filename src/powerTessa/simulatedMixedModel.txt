
rm(list=ls())

library("nlme")

pValuesTime <- vector()
pValuesDrug <- vector()
meanrSquareds <- vector()

for( z in 1:500)
{
	rSquareds <- vector()
	
	if( z== 1 ) 
	{
		pdf("c:\\temp\\sims.pdf")
		par(mfrow=c(3,2))
	}
	
	allX <- vector()
	allY <- vector()
	allPatients <- vector()
	
	# 4 drug classes evenly distributed
	# TODO: A more accurate power simulation would allow for different sized classes
	
	allDrug <- vector()
	index <- 1
	drugIndex <- 1

	#this is the sample size
	for( i in 1:200) 
	{
		x <- c(-7,0,7,14,21,28,35,42)
	
		# this controls the effect size with time
		slope <- 1
		sigma <- sqrt((slope^2 * var(x)) * (1 - 0.3) / 0.3)
		
		y <- slope * x + rnorm(length(x), sd = sigma)
		
		# this controls the effect size with drug
		if( drugIndex == 1 | drugIndex == 2 ) 
			y = y + y * 0.30
		
		rSquareds[i] =  summary(lm(y ~ x))$r.squared

		if( z == 1 ) 	
			plot(x,y, main = rSquareds[i] )
		
		for( j in 1:length(x))
		{
			allX[index] <- x[j]
			allY[index] <- y[j]
			allPatients[index] <- i
			allDrug[index] <- drugIndex		
			index <- index + 1
		}
		
		drugIndex <- drugIndex + 1
			
		if(drugIndex == 5) 
			drugIndex =1 ;
	
	}
		
	meanrSquareds[z] = mean(rSquareds)
	
	if ( z ==1 ) 
		dev.off()
	
	#x is time; y is response (tolerance)
	myFrame <- data.frame( allX, allY, allDrug, allPatients)
	
	M.mixed <- lme( allY ~ allX + allDrug, method= "REML", random = ~1 | allPatients, data = myFrame)
	
	anova(M.mixed)
	
	pValuesTime[z] <- anova(M.mixed)$"p-value"[2]
	pValuesDrug[z] <- anova(M.mixed)$"p-value"[3]
			
}


mean(meanrSquareds)

#powertime
sum(pValuesTime < 0.05  ) / length(pValuesTime )

#powerDrug 
sum(pValuesDrug < 0.05 ) / length(pValuesDrug)
