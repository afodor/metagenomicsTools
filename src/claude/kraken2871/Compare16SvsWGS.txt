# -----------------------------
# Spearman per sample helper
# -----------------------------
spearman_by_sample <- function(W, colA, colB) {
  W[!is.na(get(colA)) & !is.na(get(colB)),
    .(n_genus = .N,
      spearman = suppressWarnings(cor(get(colA), get(colB), method = "spearman"))),
    by = sampleID]
}

sp_2v1 <- spearman_by_sample(W, type2, type1)
sp_3v1 <- spearman_by_sample(W, type3, type1)
sp_4v1 <- spearman_by_sample(W, type4, type1)

# Console summaries
cat("Spearman summary (2 vs 1):\n"); print(summary(sp_2v1$spearman))
cat("\nSpearman summary (3 vs 1):\n"); print(summary(sp_3v1$spearman))
cat("\nSpearman summary (4 vs 1):\n"); print(summary(sp_4v1$spearman))

# -----------------------------
# Paired Wilcoxon tests (pairwise among 3 comparisons)
# Tests are on per-sample spearman correlations.
# Safe: returns NA if too few paired samples.
# -----------------------------
paired_wilcox_safe <- function(A, B, labelA, labelB) {
  common <- intersect(A$sampleID, B$sampleID)
  if (length(common) < 2) {
    return(list(p = NA_real_, n = length(common), label = paste0(labelA, " vs ", labelB)))
  }
  a <- A[sampleID %in% common][order(sampleID)]
  b <- B[sampleID %in% common][order(sampleID)]
  stopifnot(all(a$sampleID == b$sampleID))

  res <- wilcox.test(a$spearman, b$spearman, paired = TRUE, exact = FALSE)
  list(p = res$p.value, n = length(common), label = paste0(labelA, " vs ", labelB), res = res)
}

lab_2v1 <- paste(type2, "vs", type1)
lab_3v1 <- paste(type3, "vs", type1)
lab_4v1 <- paste(type4, "vs", type1)

# -----------------------------
# Helper: draw significance brackets + p-values on a boxplot (base R)
# -----------------------------
p_to_label <- function(p, n) {
  if (is.na(p)) return(sprintf("p=NA (n=%d)", n))
  sprintf("p=%.3g (n=%d)", p, n)
}

add_bracket <- function(x1, x2, y, h, label, cex = 0.85) {
  # bracket: up, across, down
  segments(x1, y,  x1, y + h)
  segments(x1, y + h, x2, y + h)
  segments(x2, y + h, x2, y)
  text((x1 + x2) / 2, y + h, label, pos = 3, cex = cex)
}

plot_pair <- function(Ws, colX, colY, label, sid) {
  ok <- !is.na(Ws[[colX]]) & !is.na(Ws[[colY]])
  x <- Ws[[colX]][ok]
  y <- Ws[[colY]][ok]

  plot(log10(x + 1), log10(y + 1),
       xlab = paste0("log10(", colX, " + 1)"),
       ylab = paste0("log10(", colY, " + 1)"),
       main = paste0("Sample ", sid, " (", label, "), n=", length(x)))

  if (length(x) >= 2) {
    r <- suppressWarnings(cor(x, y, method = "spearman"))
    mtext(sprintf("Spearman = %.3f", r), side = 3, line = 0.2, cex = 0.85)
  } else {
    mtext("Spearman = NA (too few points)", side = 3, line = 0.2, cex = 0.85)
  }
}

plot_one_sample <- function(sid) {
  Ws <- W[sampleID == sid]

  plot_pair(Ws, type2, type1, "2 vs 1", sid)
  plot_pair(Ws, type3, type1, "3 vs 1", sid)
  plot_pair(Ws, type4, type1, "4 vs 1", sid)
}

    plot_one_sample(samples_to_plot[i + 1])

        outer = TRUE, cex = 1.0)

  i <- i + 2
}
