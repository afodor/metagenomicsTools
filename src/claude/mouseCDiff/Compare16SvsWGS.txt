library(data.table)

setwd("C:\\claudeSummary\\simonCrossRho\\kraken_max1")

DT <- fread("allGenusFlat.txt", sep = "\t", header = TRUE)

firstID  <- 3
secondID <- 1

sampleTypes <- unique(DT$SampleType)
typeA <- sampleTypes[firstID]
typeB <- sampleTypes[secondID]

# Keep only the two types of interest
DT2 <- DT[SampleType %chin% c(typeA, typeB),
          .(sampleID, genus, SampleType, genusSum)]

# If you truly expect exactly 1 value per (sampleID, SampleType, genus), enforce it
dup <- DT2[, .N, by = .(sampleID, SampleType, genus)][N != 1]
if (nrow(dup) > 0) {
  print(dup[1:min(.N, 20)])
  stop("Logic error: not exactly 1 row per (sampleID, SampleType, genus).")
}

# Wide: one row per (sampleID, genus), columns are the two sample types
W <- dcast(DT2, sampleID + genus ~ SampleType, value.var = "genusSum")

# Compute per-sample Spearman, using only genera present in BOTH types
spearmans <- W[!is.na(get(typeA)) & !is.na(get(typeB)),
              .(spearman = cor(get(typeA), get(typeB), method = "spearman")),
              by = sampleID]

# ----- Plots (once) -----
par(mfrow = c(2, 1))
boxplot(spearmans$spearman, main = sprintf("Spearman: %s vs %s", typeA, typeB))

# Example scatter for one sample (choose the first, or change as you like)
example_sample <- spearmans$sampleID[1]
Ws <- W[sampleID == example_sample]
plot(log10(Ws[[typeA]] + 1), log10(Ws[[typeB]] + 1),
     xlab = paste0("log10(", typeA, " + 1)"),
     ylab = paste0("log10(", typeB, " + 1)"),
     main = paste("Sample:", example_sample))
