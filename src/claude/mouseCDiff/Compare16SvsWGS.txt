# -----------------------------
# Spearman per sample helper
# -----------------------------
spearman_by_sample <- function(W, colA, colB) {
  W[!is.na(get(colA)) & !is.na(get(colB)),
    .(n_genus = .N,
      spearman = suppressWarnings(cor(get(colA), get(colB), method = "spearman"))),
    by = sampleID]
}

# -----------------------------
# Paired Wilcoxon tests (pairwise among 3 comparisons)
# Tests are on per-sample spearman correlations.
# -----------------------------
paired_wilcox <- function(A, B, labelA, labelB) {
  common <- intersect(A$sampleID, B$sampleID)
  a <- A[sampleID %in% common][order(sampleID)]
  b <- B[sampleID %in% common][order(sampleID)]
  stopifnot(all(a$sampleID == b$sampleID))

  res <- wilcox.test(a$spearman, b$spearman, paired = TRUE, exact = FALSE)
  list(p = res$p.value, n = length(common), label = paste0(labelA, " vs ", labelB), res = res)
}

plot_pair <- function(Ws, colX, colY, label, sid) {
  ok <- !is.na(Ws[[colX]]) & !is.na(Ws[[colY]])
  x <- Ws[[colX]][ok]
  y <- Ws[[colY]][ok]

  plot(log10(x + 1), log10(y + 1),
       xlab = paste0("log10(", colX, " + 1)"),
       ylab = paste0("log10(", colY, " + 1)"),
       main = paste0("Sample ", sid, " (", label, "), n=", length(x)))

  if (length(x) >= 2) {
    r <- suppressWarnings(cor(x, y, method = "spearman"))
    mtext(sprintf("Spearman = %.3f", r), side = 3, line = 0.2, cex = 0.85)
  } else {
    mtext("Spearman = NA (too few points)", side = 3, line = 0.2, cex = 0.85)
  }
}

plot_one_sample <- function(sid) {

  # 3 plots across

  plot_pair(Ws, type3, type1, "3 vs 1", sid)
  plot_pair(Ws, type4, type1, "4 vs 1", sid)
}
